#include <idc.idc>

#define MAX_RENAMES 2010

static is_auto_name(name)
{
    return substr(name, 0, 4) == "sub_";
}

static main()
{
    auto ea;
    auto str;
    auto ref;
    auto func_ea;
    auto current_name;
    auto count;
    auto offset;

    count = 0;

    // walk all named items and look for string literals
    ea = next_head(0, BADADDR);

    while (ea != BADADDR && count < MAX_RENAMES)
    {
        if (is_strlit(get_full_flags(ea)))
        {
            str = get_strlit_contents(ea, -1, STRTYPE_C);

            if (str != "")
            {
                // only care about gml script strings
                if (substr(str, 0, 10) == "gml_Script")
                {
                    // skip if function with this name already exists
                    if (get_name_ea_simple(str) == BADADDR)
                    {
                        // find references to the string
                        ref = get_first_dref_to(ea);

                        while (ref != BADADDR && count < MAX_RENAMES)
                        {
                            func_ea = get_func_attr(ref, FUNCATTR_START);

                            if (func_ea != BADADDR)
                            {
                                current_name = get_func_name(func_ea);

                                // only rename autogenerated names
                                if (is_auto_name(current_name))
                                {
                                    // reference should be near function start
                                    offset = ref - func_ea;

                                    // tune threshold if needed
                                    if (offset >= 0 && offset <= 0x40)
                                    {
                                        if (set_name(func_ea, str))
                                        {
                                            msg(
                                                "Renamed %x -> %s\n",
                                                func_ea,
                                                str
                                            );

                                            count = count + 1;
                                            break;
                                        }
                                    }
                                }
                            }

                            ref = get_next_dref_to(ea, ref);
                        }
                    }
                }
            }
        }

        ea = next_head(ea, BADADDR);
    }

    msg("Done. Renamed %d functions.\n", count);
}
